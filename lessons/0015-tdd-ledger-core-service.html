Mở file types.ts vừa tạo và cùng xem các khái niệm cốt lõi nhất được mô hình hóa bằng TypeScript:

src/modules/ledger/domain/types.ts — Định nghĩa kiểu dữ liệu thuần túy
// 1. Chiều của bút toán: Chỉ có thể là Ghi Nợ (DEBIT) hoặc Ghi Có (CREDIT)
export type EntryDirection = 'DEBIT' | 'CREDIT';

// 2. Một tài khoản trong Sổ cái
export interface Account {
  id: string;
  code: string;  // Ví dụ: '112' (Tiền gửi), '511' (Doanh thu)
  name: string;
  type: 'ASSET' | 'LIABILITY' | 'EQUITY' | 'REVENUE' | 'EXPENSE';
  createdAt: Date;
}

// 3. Một dòng bút toán chi tiết (Append-Only)
export interface LedgerEntry {
  id: string;
  transactionId: string;
  accountId: string;
  amount: number;
  direction: EntryDirection;
  createdAt: Date;
}

// 4. Seam Interface: Hợp đồng lưu trữ dữ liệu (Không chứa bất kỳ chữ 'supabase' nào!)
export interface ILedgerRepository {
  createAccount(account: Omit<Account, 'id' | 'createdAt'>): Promise<Account>;
  getAccountByCode(code: string): Promise<Account | null>;
  recordTransaction(
    description: string,
    entries: { accountId: string; amount: number; direction: EntryDirection }[],
    referenceId?: string
  ): Promise<{ transaction: AccountingTransaction; entries: LedgerEntry[] }>;
  getTrialBalance(): Promise<TrialBalanceRow[]>;
}

import { describe, it, expect, beforeEach } from 'vitest';
import { InMemoryLedgerRepository } from './adapters/in-memory';
import { LedgerService, LedgerBalanceError } from './service';

describe('LedgerService Core', () => {
  let repo: InMemoryLedgerRepository;
  let service: LedgerService;

  beforeEach(() => {
    repo = new InMemoryLedgerRepository();
    service = new LedgerService(repo);
  });

  it('phải ném ra lỗi khi tổng Nợ không bằng tổng Có', async () => {
    const act = () => service.recordTransaction('Test unbalanced', [
      { accountCode: '112', amount: 100, direction: 'DEBIT' },
      { accountCode: '511', amount: 90, direction: 'CREDIT' }, // Lệch 10 đồng!
    ]);

    await expect(act).rejects.toThrow(LedgerBalanceError);
  });

4. Giai đoạn 4: Viết Code Tối Thiểu Để Xanh (GREEN) & Né Bẫy AI

Bây giờ chúng ta tạo file src/modules/ledger/service.ts để làm cho test chuyển sang màu xanh.

⚠️ Bẫy ngầm của AI: Sai số dấu phẩy động (0.1 + 0.2 === 0.30000000000000004)

Khi bảo AI so sánh hai vế Nợ và Có, AI thường viết câu lệnh ngây thơ:

if (debitSum !== creditSum) throw new LedgerBalanceError(...);

Trong JavaScript, nếu một bút toán có số tiền lẻ như 19.99 và 10.01, tổng số có thể bị sai lệch phần triệu và phép tính so sánh !== sẽ từ chối oan một giao dịch hoàn toàn hợp lệ! Lập trình viên phải yêu cầu AI bọc hàm làm tròn tiền tệ:

src/modules/ledger/service.ts — Code tối thiểu vượt qua test
export class LedgerService {
  constructor(private readonly repo: ILedgerRepository) {}

  private roundMoney(val: number): number {
    return Math.round((val + Number.EPSILON) * 100) / 100;
  }

  async recordTransaction(desc: string, entries: CreateEntryDTO[]) {
    let debitSum = 0;
    let creditSum = 0;

    for (const e of entries) {
      if (e.amount <= 0) throw new LedgerValidationError('Số tiền phải > 0');
      if (e.direction === 'DEBIT') debitSum += e.amount;
      else creditSum += e.amount;
    }

    if (this.roundMoney(debitSum) !== this.roundMoney(creditSum)) {
      throw new LedgerBalanceError('Mất cân bằng Nợ/Có');
    }

    // Lưu vào repository
    return this.repo.recordTransaction(desc, entries);
  }
}

Chạy lại lệnh kiểm thử: Test lập tức chuyển sang màu XANH (PASS)! Đó chính là bản chất của vòng lặp Red-Green.
